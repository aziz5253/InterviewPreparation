
interface Bike {
    void turnOnEngine();
    void accelerate();
}

class MotorCycle implements Bike {

    boolean isEngineOn;
    int speed;

    public void turnOnEngine() {
        // turn on the engine
        isEngineOn = true;
    }

    public void accelerate() {
        // increase the speed
        speed = speed + 10;
    }
}
