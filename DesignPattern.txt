SOLID Principle
Single responsibility principle
Class should have Single Responsibility 
There should be not more than one reason for a class to change
class UserService {
    
    public void createUser(String name) {
        // Business logic
        System.out.println("User created: " + name);

        // Saving to DB (Persistence Responsibility)
        saveToDatabase(name);

        // Sending notification (Another responsibility)
        sendEmail(name);
    }

    private void saveToDatabase(String name) {
        System.out.println("Saved to DB");
    }

    private void sendEmail(String name) {
        System.out.println("Email sent");
    }
}
DB change → modify class
Email logic change → modify class
Business rule change → modify class

@Component
public class PaymentFactory {

    private Map<String, PaymentProcessor> processors;

    public PaymentFactory(List<PaymentProcessor> processorList) {
        processors = new HashMap<>();
        processorList.forEach(p -> 
            processors.put(p.getClass().getSimpleName(), p));
    }

    public PaymentProcessor getProcessor(String type) {
        return processors.get(type);
    }
}

//Builder
package org.example.builder;

public class User {

    private final String name;
    private final String email;
    private final String phone;
    private final String address;

    private User(Builder builder) {
        this.name = builder.name;
        this.email = builder.email;
        this.phone = builder.phone;
        this.address = builder.address;
    }

    public static class Builder {

        // Mandatory field
        private final String name;

        // Optional fields
        private String email;
        private String phone;
        private String address;

        public Builder(String name) {
            this.name = name;
        }

        public Builder email(String email) {
            this.email = email;
            return this;
        }

        public Builder phone(String phone) {
            this.phone = phone;
            return this;
        }

        public Builder address(String address) {
            this.address = address;
            return this;
        }

        public User build() {
            return new User(this);
        }
    }
}
